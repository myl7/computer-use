% Dual-model table cell: the GLM value on blue, the DeepSeek value on amber.
\newcommand{\dual}[2]{{\setlength{\fboxsep}{1.25pt}\colorbox{glmTint}{#1}\hspace{1pt}\colorbox{dsTint}{#2}}}
% Single-model cell: the color names the model; the other model has no data.
\newcommand{\dualG}[1]{{\setlength{\fboxsep}{1.25pt}\colorbox{glmTint}{#1}}}
\newcommand{\dualD}[1]{{\setlength{\fboxsep}{1.25pt}\colorbox{dsTint}{#1}}}
% Three-model cell: GLM on blue, DeepSeek on amber, Qwen on mint.
\newcommand{\tri}[3]{{\setlength{\fboxsep}{1.25pt}\colorbox{glmTint}{#1}\hspace{1pt}\colorbox{dsTint}{#2}\hspace{1pt}\colorbox{qwTint}{#3}}}
% Single-model cells inside a three-model table.
\newcommand{\triG}[1]{{\setlength{\fboxsep}{1.25pt}\colorbox{glmTint}{#1}}}
\newcommand{\triD}[1]{{\setlength{\fboxsep}{1.25pt}\colorbox{dsTint}{#1}}}
\newcommand{\triQ}[1]{{\setlength{\fboxsep}{1.25pt}\colorbox{qwTint}{#1}}}
% Two-of-three cells: the absent model is the missing color.
\newcommand{\triGQ}[2]{{\setlength{\fboxsep}{1.25pt}\colorbox{glmTint}{#1}\hspace{1pt}\colorbox{qwTint}{#2}}}
\newcommand{\triDQ}[2]{{\setlength{\fboxsep}{1.25pt}\colorbox{dsTint}{#1}\hspace{1pt}\colorbox{qwTint}{#2}}}
% Vertically stacked three-model cell (Table 2): one colored line per model,
% top to bottom GLM, DeepSeek, Qwen; a missing color marks a model with no
% value in that column.
\newcommand{\vstack}[1]{\begin{tabular}[c]{@{}r@{}}#1\end{tabular}}

\newcommand{\vdual}[2]{\vstack{\cellcolor{glmTint}#1\\\cellcolor{dsTint}#2}}
\newcommand{\vtriGQ}[2]{\vstack{\cellcolor{glmTint}#1\\\cellcolor{qwTint}#2}}
\newcommand{\vtriDQ}[2]{\vstack{\cellcolor{dsTint}#1\\\cellcolor{qwTint}#2}}
\newcommand{\vtriG}[1]{\vstack{\cellcolor{glmTint}#1}}
\newcommand{\vtriD}[1]{\vstack{\cellcolor{dsTint}#1}}
\newcommand{\vtriQ}[1]{\vstack{\cellcolor{qwTint}#1}}
